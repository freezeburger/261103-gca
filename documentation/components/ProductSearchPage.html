import { CurrencyPipe } from '@angular/common';
import { Component, computed, inject, input, numberAttribute } from '@angular/core';
import { rxResource } from '@angular/core/rxjs-interop';
import { ActivatedRoute, Params, Router, RouterLink } from '@angular/router';
import {
  GcaButton,
  GcaCard,
  GcaEmptyState,
  GcaImage,
  GcaPageEvent,
  GcaPaginator,
  GcaSearchInput,
  GcaSelect,
  GcaSkeleton,
  GcaSort,
  GcaSortHeader,
  GcaTable,
} from 'gca';
import { ProductApi, SortOrder } from '../product-api';

const DEFAULT_SIZE = 20;
const SORT_OPTIONS = [
  { value: '', label: 'Pertinence' },
  { value: 'title:asc', label: 'Titre A → Z' },
  { value: 'title:desc', label: 'Titre Z → A' },
  { value: 'price:asc', label: 'Prix croissant' },
  { value: 'price:desc', label: 'Prix décroissant' },
] as const;

@Component({
  selector: 'app-product-search-page',
  imports: [
    CurrencyPipe,
    RouterLink,
    GcaButton,
    GcaCard,
    GcaEmptyState,
    GcaImage,
    GcaPaginator,
    GcaSearchInput,
    GcaSelect,
    GcaSkeleton,
    GcaSortHeader,
    GcaTable,
  ],
  templateUrl: './product-search-page.html',
  styleUrl: './product-search-page.scss',
})
export default class ProductSearchPage {
  private readonly api = inject(ProductApi);
  private readonly router = inject(Router);
  private readonly route = inject(ActivatedRoute);

  // Paramètres d'URL reçus via withComponentInputBinding()
  readonly q = input('', { transform: (value: string | undefined) => value ?? '' });
  readonly page = input(1, { transform: (value: unknown) => Math.max(1, numberAttribute(value, 1)) });
  readonly size = input(DEFAULT_SIZE, { transform: (value: unknown) => numberAttribute(value, DEFAULT_SIZE) });
  readonly sortBy = input<string>();
  readonly order = input<SortOrder>();

  protected readonly sortOptions = SORT_OPTIONS;
  protected readonly skeletonRows = [1, 2, 3, 4, 5, 6];
  protected readonly sort = computed<GcaSort | null>(() => {
    const field = this.sortBy();
    return field ? { field, order: this.order() ?? 'asc' } : null;
  });
  protected readonly sortKey = computed(() => {
    const sort = this.sort();
    return sort ? `${sort.field}:${sort.order}` : '';
  });

  protected readonly products = rxResource({
    params: () => ({ q: this.q(), page: this.page(), size: this.size(), sortBy: this.sortBy(), order: this.order() }),
    stream: ({ params }) => this.api.search(params),
  });

  protected navigate(queryParams: Params): void {
    void this.router.navigate([], { relativeTo: this.route, queryParams, queryParamsHandling: 'merge' });
  }

  protected search(q: string): void {
    this.navigate({ q: q.trim() || null, page: null });
  }

  protected changeSort(sort: GcaSort | null): void {
    this.navigate({ sortBy: sort?.field ?? null, order: sort?.order ?? null, page: null });
  }

  protected changeSortKey(event: Event): void {
    const [field, order] = (event.target as HTMLSelectElement).value.split(':');
    this.changeSort(field ? { field, order: order === 'desc' ? 'desc' : 'asc' } : null);
  }

  protected changePage({ page, pageSize }: GcaPageEvent): void {
    this.navigate({ page: page > 1 ? page : null, size: pageSize === DEFAULT_SIZE ? null : pageSize });
  }

  protected reset(): void {
    void this.router.navigate([], { relativeTo: this.route });
  }

  protected open(id: number): void {
    void this.router.navigate([id], { relativeTo: this.route, queryParamsHandling: 'preserve' });
  }
}

<gca-card>
  <div class="toolbar" role="search">
    <gca-search-input class="search" placeholder="Rechercher un produit" [value]="q()" (search)="search($event)" />
    <select gcaSelect class="sort" aria-label="Trier par" (change)="changeSortKey($event)">
      @for (option of sortOptions; track option.value) {
        <option [value]="option.value" [selected]="option.value === sortKey()">{{ option.label }}</option>
      }
    </select>
    <button gcaButton type="button" variant="text" severity="secondary" (click)="reset()">Réinitialiser</button>
  </div>

  @if (products.error()) {
    <gca-empty-state icon="⚠" heading="Impossible de charger les produits." description="Vos filtres sont conservés.">
      <button gcaButton type="button" icon="↻" (click)="products.reload()">Réessayer</button>
    </gca-empty-state>
  } @else if (products.hasValue()) {
    @let result = products.value();
    @if (result.products.length === 0) {
      <gca-empty-state
        icon="⌕"
        heading="Aucun résultat"
        [description]="q() ? 'Aucun produit ne correspond à « ' + q() + ' ».' : 'Le catalogue est vide.'"
      >
        <button gcaButton type="button" variant="outlined" (click)="reset()">Réinitialiser</button>
      </gca-empty-state>
    } @else {
      <p class="summary" aria-live="polite">
        <strong>{{ result.total }}</strong> résultat{{ result.total > 1 ? 's' : '' }}
        @if (q()) {
          pour « {{ q() }} »
        }
      </p>
      <div class="table-scroll">
        <table gcaTable [sort]="sort()" [loading]="products.isLoading()" (sortChange)="changeSort($event)">
          <thead>
            <tr>
              <th><span class="sr-only">Miniature</span></th>
              <th>ID</th>
              <th gcaSortHeader="title">Titre</th>
              <th>Description</th>
              <th gcaSortHeader="price" class="gca-num">Prix</th>
              <th class="gca-actions">Actions</th>
            </tr>
          </thead>
          <tbody>
            @for (product of result.products; track product.id) {
              <tr class="gca-row-link" tabindex="0" (click)="open(product.id)" (keydown.enter)="open(product.id)">
                <td>
                  <gca-image compact width="36px" height="36px" [src]="product.thumbnail" [alt]="product.title" />
                </td>
                <td>{{ product.id }}</td>
                <td><strong>{{ product.title }}</strong></td>
                <td class="gca-truncate" [title]="product.description">{{ product.description }}</td>
                <td class="gca-num">{{ product.price | currency: 'EUR' }}</td>
                <td class="gca-actions" (click)="$event.stopPropagation()" (keydown.enter)="$event.stopPropagation()">
                  <a gcaButton size="sm" variant="text" queryParamsHandling="preserve" [routerLink]="[product.id]">Voir</a>
                  <a
                    gcaButton
                    size="sm"
                    variant="outlined"
                    queryParamsHandling="preserve"
                    [routerLink]="[product.id, 'edit']"
                  >
                    Modifier
                  </a>
                </td>
              </tr>
            }
          </tbody>
        </table>
      </div>
      <gca-paginator [total]="result.total" [page]="page()" [pageSize]="size()" (paginate)="changePage($event)" />
    }
  } @else {
    <div class="skeletons" aria-busy="true" aria-label="Chargement des produits">
      @for (row of skeletonRows; track row) {
        <div class="skeleton-row">
          <gca-skeleton shape="circle" width="36px" />
          <gca-skeleton width="25%" />
          <gca-skeleton width="45%" />
          <gca-skeleton width="10%" />
        </div>
      }
    </div>
  }
